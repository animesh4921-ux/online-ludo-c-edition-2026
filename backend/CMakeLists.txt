cmake_minimum_required(VERSION 3.16)
project(online_ludo_c C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(ludo_engine src/ludo.c)
target_include_directories(ludo_engine PUBLIC include)
add_executable(ludo_server src/server.c)
target_link_libraries(ludo_server PRIVATE ludo_engine)
if(WIN32)
  target_link_libraries(ludo_server PRIVATE ws2_32)
endif()
add_executable(ludo_engine_test test/test_ludo.c)
target_link_libraries(ludo_engine_test PRIVATE ludo_engine)
enable_testing()
add_test(NAME ludo_engine_test COMMAND ludo_engine_test)
